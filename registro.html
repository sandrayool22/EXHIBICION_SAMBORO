function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. ACCIÓN: REGISTRO MASIVO DE SUBFAMILIAS if (data.action === "guardarBitacoraMasiva") { var sheetBitacora = ss.getSheetByName("BITACORA") || ss.insertSheet("BITACORA"); // Encabezados con Criterios de Exhibición integrados if (sheetBitacora.getLastRow() === 0) { sheetBitacora.appendRow([ "Fecha / Hora", "Usuario", "Correo", "Tienda", "Subfamilia", "Exhibidor Limpio (C1)", "Precios Visibles (C2)", "Muestras Completas (C3)", "Observación / Hallazgo", "Imagen_URL" ]); } var folderName = "Fotos_Bitacora_Samboro"; var folder = DriveApp.getFoldersByName(folderName); var targetFolder = folder.hasNext() ? folder.next() : DriveApp.createFolder(folderName); // Recorrer y guardar subfamilia por subfamilia data.evaluaciones.forEach(function(item) { var imagenUrl = ""; if (item.fotoBase64) { var blob = Utilities.newBlob( Utilities.base64Decode(item.fotoBase64), "image/jpeg", "Foto_" + data.usuario + "_" + item.subfamilia.replace(/ /g, "_") + "_" + new Date().getTime() + ".jpg" ); var file = targetFolder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); imagenUrl = file.getUrl(); } sheetBitacora.appendRow([ new Date(), data.usuario, data.correo, data.tienda, item.subfamilia, item.criterio1, item.criterio2, item.criterio3, item.observacion, imagenUrl ]); }); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Todas las subfamilias registradas." })) .setMimeType(ContentService.MimeType.JSON); } // 2. ACCIÓN: LOGIN DE USUARIOS var codigoIngresado = String(data.codigo).trim().toUpperCase(); var sheetAsesores = ss.getSheetByName("ASESORES"); var datos = sheetAsesores.getDataRange().getValues(); for (var i = 1; i < datos.length; i++) { var codigoBD = String(datos[i][0]).trim().toUpperCase(); if (codigoBD === codigoIngresado) { var respuesta = { success: true, usuario: { codigo: datos[i][0], correo: datos[i][1], rol: datos[i][2], tienda: datos[i][3], subfamilias: datos[i][4] } }; return ContentService.createTextOutput(JSON.stringify(respuesta)) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Código no encontrado." })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Error: " + error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
